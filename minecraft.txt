    <script>
        /* ==========================================================================
           1. GRAPHICS ENGINE & ATMOSPHERIC LIGHTING SETUP
           ========================================================================== */
        const canvas = document.getElementById('game-canvas');
        const scene = new THREE.Scene();
        
        scene.background = new THREE.Color(0x0a182e);
        scene.fog = new THREE.FogExp2(0x0a182e, 0.007);

        const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
        scene.add(camera);

        const renderer = new THREE.WebGLRenderer({ canvas: canvas, antialias: true, powerPreference: "high-performance" });
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
        renderer.shadowMap.enabled = true;
        renderer.shadowMap.type = THREE.PCFSoftShadowMap;
        renderer.toneMapping = THREE.ACESFilmicToneMapping;
        renderer.toneMappingExposure = 1.25;

        // Ambient & Sun Lighting
        const ambientLight = new THREE.AmbientLight(0xddeeff, 0.55);
        scene.add(ambientLight);

        const sunLight = new THREE.DirectionalLight(0xfffaed, 1.3);
        sunLight.position.set(80, 130, 60);
        sunLight.castShadow = true;
        sunLight.shadow.mapSize.width = 2048;
        sunLight.shadow.mapSize.height = 2048;
        sunLight.shadow.camera.near = 0.5;
        sunLight.shadow.camera.far = 300;
        scene.add(sunLight);

        const hemiLight = new THREE.HemisphereLight(0xffffff, 0x444455, 0.45);
        scene.add(hemiLight);

        /* ==========================================================================
           2. CONTROLS & CAMERA SYSTEM
           ========================================================================== */
        const controls = new THREE.PointerLockControls(camera, document.body);
        const chatInput = document.getElementById('chat-input');

        document.body.addEventListener('click', (e) => {
            if (document.activeElement !== chatInput) {
                controls.lock();
            }
        });

        /* ==========================================================================
           3. WORLD GENERATION & TEXTURED BLOCK MATERIALS
           ========================================================================== */
        const blockGeometry = new THREE.BoxGeometry(1, 1, 1);
        const materials = {
            grass: new THREE.MeshStandardMaterial({ color: 0x2e7d32, roughness: 0.8 }),
            dirt: new THREE.MeshStandardMaterial({ color: 0x5d4037, roughness: 0.9 }),
            stone: new THREE.MeshStandardMaterial({ color: 0x616161, roughness: 0.6 }),
            wood: new THREE.MeshStandardMaterial({ color: 0x4e342e, roughness: 0.7 }),
            gold: new THREE.MeshStandardMaterial({ color: 0xffd700, metalness: 0.85, roughness: 0.25 }),
            bedrock: new THREE.MeshStandardMaterial({ color: 0x111111, roughness: 0.95 })
        };

        const worldBlocks = new Map();

        function createBlock(x, y, z, type) {
            const key = `${Math.floor(x)},${Math.floor(y)},${Math.floor(z)}`;
            if (worldBlocks.has(key)) return;

            const mesh = new THREE.Mesh(blockGeometry, materials[type] || materials.dirt);
            mesh.position.set(x, y, z);
            mesh.castShadow = true;
            mesh.receiveShadow = true;

            scene.add(mesh);
            worldBlocks.set(key, mesh);
            return mesh;
        }

        function removeBlock(x, y, z) {
            const key = `${Math.floor(x)},${Math.floor(y)},${Math.floor(z)}`;
            if (worldBlocks.has(key)) {
                const mesh = worldBlocks.get(key);
                scene.remove(mesh);
                mesh.geometry.dispose();
                worldBlocks.delete(key);
            }
        }

        // Generate Terrain
        function generateTerrain() {
            for (let x = -20; x <= 20; x++) {
                for (let z = -20; z <= 20; z++) {
                    createBlock(x, 0, z, 'bedrock');
                    for (let y = 1; y <= 30; y++) createBlock(x, y, z, 'stone');
                    for (let y = 31; y <= 33; y++) createBlock(x, y, z, 'dirt');
                    createBlock(x, 34, z, 'grass');
                }
            }
        }
        generateTerrain();

        /* ==========================================================================
           4. MALE LEADER OUTFIT & WEAPONS MODELING
           ========================================================================== */
        const playerMaleGroup = new THREE.Group();
        
        // Armored Leather Jacket
        const torsoMesh = new THREE.Mesh(
            new THREE.BoxGeometry(0.6, 0.8, 0.4),
            new THREE.MeshStandardMaterial({ color: 0x1a1a1a, roughness: 0.4, metalness: 0.3 })
        );
        torsoMesh.position.y = -0.6;
        playerMaleGroup.add(torsoMesh);

        // Gold Leader Badge on Chest
        const badgeMesh = new THREE.Mesh(
            new THREE.CylinderGeometry(0.06, 0.06, 0.02, 6),
            new THREE.MeshStandardMaterial({ color: 0xFFD700, metalness: 0.9, roughness: 0.1 })
        );
        badgeMesh.rotation.x = Math.PI / 2;
        badgeMesh.position.set(0.18, -0.45, -0.21);
        playerMaleGroup.add(badgeMesh);

        camera.add(playerMaleGroup);

        // --- EXCLUSIVE MASTER WAND ---
        const wandGroup = new THREE.Group();
        const wandShaft = new THREE.Mesh(
            new THREE.CylinderGeometry(0.012, 0.02, 0.75, 12),
            new THREE.MeshStandardMaterial({ color: 0x1a0f0a, roughness: 0.3, metalness: 0.2 })
        );
        wandGroup.add(wandShaft);

        const wandTip = new THREE.Mesh(
            new THREE.OctahedronGeometry(0.09, 2),
            new THREE.MeshStandardMaterial({ color: 0x00FFFF, emissive: 0x00FFFF, emissiveIntensity: 1.5, roughness: 0.1 })
        );
        wandTip.position.y = 0.4;
        wandGroup.add(wandTip);

        const wandLight = new THREE.PointLight(0x00FFFF, 2, 8);
        wandTip.add(wandLight);

        wandGroup.position.set(0.38, -0.28, -0.5);
        wandGroup.rotation.x = Math.PI / 5;
        camera.add(wandGroup);

        // --- HEAVY TACTICAL RIFLE ---
        const gunGroup = new THREE.Group();
        const gunBody = new THREE.Mesh(
            new THREE.BoxGeometry(0.09, 0.14, 0.55),
            new THREE.MeshStandardMaterial({ color: 0x111111, metalness: 0.9, roughness: 0.2 })
        );
        gunGroup.add(gunBody);

        const gunBarrel = new THREE.Mesh(
            new THREE.CylinderGeometry(0.022, 0.022, 0.4, 12),
            new THREE.MeshStandardMaterial({ color: 0x222222, metalness: 0.95 })
        );
        gunBarrel.rotation.x = Math.PI / 2;
        gunBarrel.position.set(0, 0.03, -0.38);
        gunGroup.add(gunBarrel);

        const muzzleFlash = new THREE.PointLight(0xFF6600, 0, 10);
        muzzleFlash.position.set(0, 0.03, -0.6);
        gunGroup.add(muzzleFlash);

        gunGroup.position.set(0.38, -0.28, -0.5);
        gunGroup.visible = false;
        camera.add(gunGroup);

        /* ==========================================================================
           5. SPELLS & PARTICLE FX SYSTEM
           ========================================================================== */
        const activeLightnings = [];
        const activeFireballs = [];

        function castLightning(targetPos) {
            const points = [];
            let current = new THREE.Vector3(targetPos.x, targetPos.y + 40, targetPos.z);
            points.push(current.clone());

            while (current.y > targetPos.y) {
                current.y -= 1.8;
                current.x += (Math.random() - 0.5) * 1.5;
                current.z += (Math.random() - 0.5) * 1.5;
                points.push(current.clone());
            }

            const geometry = new THREE.BufferGeometry().setFromPoints(points);
            const material = new THREE.LineBasicMaterial({ color: 0x00FFFF, linewidth: 4 });
            const lightningLine = new THREE.Line(geometry, material);
            scene.add(lightningLine);

            const flashLight = new THREE.PointLight(0x00FFFF, 6, 25);
            flashLight.position.set(targetPos.x, targetPos.y + 2, targetPos.z);
            scene.add(flashLight);

            activeLightnings.push({ line: lightningLine, light: flashLight, age: 0 });
        }

        function castFireball() {
            const fireball = new THREE.Mesh(
                new THREE.SphereGeometry(0.28, 16, 16),
                new THREE.MeshBasicMaterial({ color: 0xFF3300 })
            );

            const camPos = new THREE.Vector3();
            camera.getWorldPosition(camPos);
            const camDir = new THREE.Vector3();
            camera.getWorldDirection(camDir);

            fireball.position.copy(camPos).add(camDir.clone().multiplyScalar(0.9));
            scene.add(fireball);

            const fireLight = new THREE.PointLight(0xFF4500, 4, 12);
            fireball.add(fireLight);

            activeFireballs.push({ mesh: fireball, direction: camDir, speed: 40.0, age: 0 });
        }

        /* ==========================================================================
           6. INPUT & CHAT SYSTEM CONTROLS
           ========================================================================== */
        camera.position.set(0, 36, 4);
        let moveForward = false, moveBackward = false, moveLeft = false, moveRight = false;
        let isFlying = true;
        let velocity = new THREE.Vector3();
        let activeWeapon = 'wand';
        let selectedBlockType = 'stone';

        // Chat Box Hotkey 'T' & Message Broadcasting
        document.addEventListener('keydown', (e) => {
            if (document.activeElement === chatInput) {
                if (e.key === 'Enter' && chatInput.value.trim() !== '') {
                    const msgContainer = document.getElementById('chat-messages');
                    const newMsg = document.createElement('div');
                    newMsg.innerHTML = `<b style="color: #FFD700;">👑 YOU (Leader):</b> ${chatInput.value}`;
                    msgContainer.appendChild(newMsg);
                    msgContainer.scrollTop = msgContainer.scrollHeight;
                    chatInput.value = '';
                    chatInput.blur();
                    controls.lock();
                }
                return;
            }

            switch (e.code) {
                case 'KeyT':
                    e.preventDefault();
                    controls.unlock();
                    chatInput.focus();
                    break;
                case 'KeyW': moveForward = true; break;
                case 'KeyS': moveBackward = true; break;
                case 'KeyA': moveLeft = true; break;
                case 'KeyD': moveRight = true; break;
                case 'Space': if (isFlying) velocity.y = 0.3; break;
                case 'ShiftLeft': if (isFlying) velocity.y = -0.3; break;
                case 'KeyG': isFlying = !isFlying; break;
                
                case 'KeyE':
                    if (activeWeapon === 'wand') {
                        activeWeapon = 'gun';
                        wandGroup.visible = false;
                        gunGroup.visible = true;
                        document.getElementById('weapon-status').innerText = "ACTIVE WEAPON: HEAVY TACTICAL RIFLE [Press E to Switch]";
                        document.getElementById('slot-wand').classList.remove('active');
                        document.getElementById('slot-gun').classList.add('active');
                    } else {
                        activeWeapon = 'wand';
                        wandGroup.visible = true;
                        gunGroup.visible = false;
                        document.getElementById('weapon-status').innerText = "ACTIVE WEAPON: EXCLUSIVE MASTER WAND [Press E to Switch]";
                        document.getElementById('slot-gun').classList.remove('active');
                        document.getElementById('slot-wand').classList.add('active');
                    }
                    break;

                case 'Digit3': selectedBlockType = 'stone'; break;
                case 'Digit4': selectedBlockType = 'dirt'; break;
                case 'Digit5': selectedBlockType = 'gold'; break;
            }
        });

        document.addEventListener('keyup', (e) => {
            if (document.activeElement === chatInput) return;
            switch (e.code) {
                case 'KeyW': moveForward = false; break;
                case 'KeyS': moveBackward = false; break;
                case 'KeyA': moveLeft = false; break;
                case 'KeyD': moveRight = false; break;
                case 'Space': case 'ShiftLeft': if (isFlying) velocity.y = 0; break;
            }
        });

        /* ==========================================================================
           7. SHOOTING & BLOCK INTERACTION
           ========================================================================== */
        const raycaster = new THREE.Raycaster();
        const center = new THREE.Vector2(0, 0);

        window.addEventListener('mousedown', (e) => {
            if (!controls.isLocked || document.activeElement === chatInput) return;

            if (activeWeapon === 'gun') {
                if (e.button === 0) {
                    muzzleFlash.intensity = 10;
                    setTimeout(() => muzzleFlash.intensity = 0, 40);

                    gunGroup.position.z += 0.06;
                    setTimeout(() => gunGroup.position.z -= 0.06, 60);

                    raycaster.setFromCamera(center, camera);
                    const intersects = raycaster.intersectObjects(scene.children);

                    if (intersects.length > 0) {
                        const hit = intersects[0];
                        if (hit.object !== torsoMesh && hit.object !== badgeMesh && hit.object !== gunBody) {
                            const pos = hit.object.position;
                            removeBlock(pos.x, pos.y, pos.z);
                        }
                    }
                }
            } 
            else if (activeWeapon === 'wand') {
                raycaster.setFromCamera(center, camera);
                const intersects = raycaster.intersectObjects(scene.children);

                if (intersects.length > 0) {
                    let hit = null;
                    for (let i = 0; i < intersects.length; i++) {
                        if (intersects[i].object !== wandShaft && intersects[i].object !== wandTip && intersects[i].object !== torsoMesh) {
                            hit = intersects[i];
                            break;
                        }
                    }
                    if (!hit) return;

                    if (e.button === 0) {
                        const pos = hit.object.position;
                        castLightning(pos);
                        removeBlock(pos.x, pos.y, pos.z);
                    } else if (e.button === 2) {
                        castFireball();
                        const normal = hit.face.normal;
                        const pos = hit.object.position;
                        createBlock(pos.x + normal.x, pos.y + normal.y, pos.z + normal.z, selectedBlockType);
                    }
                }
            }
        });

        /* ==========================================================================
           8. ANIMATION & RENDER LOOP
           ========================================================================== */
        let prevTime = performance.now();

        function animate() {
            requestAnimationFrame(animate);

            const time = performance.now();
            const delta = (time - prevTime) / 1000;

            if (controls.isLocked) {
                velocity.x -= velocity.x * 10.0 * delta;
                velocity.z -= velocity.z * 10.0 * delta;

                let dirZ = Number(moveForward) - Number(moveBackward);
                let dirX = Number(moveRight) - Number(moveLeft);

                if (moveForward || moveBackward) velocity.z -= dirZ * 50.0 * delta;
                if (moveLeft || moveRight) velocity.x -= dirX * 50.0 * delta;

                controls.moveForward(-velocity.z * delta);
                controls.moveRight(-velocity.x * delta);

                if (isFlying) camera.position.y += velocity.y;

                wandTip.rotation.y += 0.06;
                wandTip.rotation.z += 0.03;
            }

            for (let i = activeLightnings.length - 1; i >= 0; i--) {
                const l = activeLightnings[i];
                l.age += delta;
                if (l.age > 0.12) {
                    scene.remove(l.line);
                    scene.remove(l.light);
                    activeLightnings.splice(i, 1);
                }
            }

            for (let i = activeFireballs.length - 1; i >= 0; i--) {
                const f = activeFireballs[i];
                f.mesh.position.add(f.direction.clone().multiplyScalar(f.speed * delta));
                f.age += delta;

                if (f.age > 2.5) {
                    scene.remove(f.mesh);
                    activeFireballs.splice(i, 1);
                }
            }

            prevTime = time;
            renderer.render(scene, camera);
        }

        animate();

        window.addEventListener('resize', () => {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        });
    </script>
